# Raft Project
cmake_minimum_required(VERSION 3.16)

# Determine the RaftCore version pinned in features.cmake (single source of truth)
# RaftCore@<tag> -> bootstrap from that release's assets (fully pinned build)
# RaftCore@main -> bootstrap from the latest release (build floats by choice)
set(_raft_core_tag "main")
file(GLOB _raft_feature_files "${CMAKE_SOURCE_DIR}/systypes/*/features.cmake")
foreach(_raft_ff ${_raft_feature_files})
    file(STRINGS "${_raft_ff}" _raft_core_lines REGEX "RaftCore[@#]")
    if(_raft_core_lines)
        list(GET _raft_core_lines 0 _raft_core_line)
        string(REGEX MATCH "RaftCore[@#]([^ \t\r\n\")]+)" _raft_unused "${_raft_core_line}")
        if(CMAKE_MATCH_1)
            set(_raft_core_tag "${CMAKE_MATCH_1}")
        endif()
        break()
    endif()
endforeach()
if(_raft_core_tag STREQUAL "main")
    set(BOOTSTRAP_URL "https://github.com/robdobsn/RaftCore/releases/latest/download/RaftBootstrap.cmake")
else()
    set(BOOTSTRAP_URL "https://github.com/robdobsn/RaftCore/releases/download/${_raft_core_tag}/RaftBootstrap.cmake")
endif()

# Bootstrap using the RaftBootstrap script
# A local RaftCore copy in raftdevlibs takes precedence (development workflow)
if(EXISTS "${CMAKE_SOURCE_DIR}/raftdevlibs/RaftCore/scripts/RaftBootstrap.cmake")
    include("${CMAKE_SOURCE_DIR}/raftdevlibs/RaftCore/scripts/RaftBootstrap.cmake")
else()
    # Download only when missing or the pinned version changed; a failed
    # download must never break a previously-working build
    set(_raft_bs_file "${CMAKE_BINARY_DIR}/RaftBootstrap.cmake")
    set(_raft_bs_stamp "${CMAKE_BINARY_DIR}/RaftBootstrap.url")
    set(_raft_bs_prev_url "")
    if(EXISTS "${_raft_bs_stamp}")
        file(READ "${_raft_bs_stamp}" _raft_bs_prev_url)
    endif()
    if(NOT EXISTS "${_raft_bs_file}" OR NOT "${_raft_bs_prev_url}" STREQUAL "${BOOTSTRAP_URL}")
        file(DOWNLOAD "${BOOTSTRAP_URL}" "${_raft_bs_file}.tmp" STATUS _raft_bs_status)
        list(GET _raft_bs_status 0 _raft_bs_code)
        if(_raft_bs_code EQUAL 0)
            file(RENAME "${_raft_bs_file}.tmp" "${_raft_bs_file}")
            file(WRITE "${_raft_bs_stamp}" "${BOOTSTRAP_URL}")
        else()
            file(REMOVE "${_raft_bs_file}.tmp")
            if(EXISTS "${_raft_bs_file}")
                message(WARNING "RaftBootstrap.cmake download failed (${BOOTSTRAP_URL}) - using previously downloaded copy")
            else()
                message(FATAL_ERROR "Failed to download RaftBootstrap.cmake from ${BOOTSTRAP_URL}")
            endif()
        endif()
    endif()
    include("${_raft_bs_file}")
endif()

# Define the project dependencies
project(${_systype_name} DEPENDS ${ADDED_PROJECT_DEPENDENCIES})
